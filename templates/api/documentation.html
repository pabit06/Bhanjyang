Usage Examples

Performance API

# Get page load performance data
curl "{{ request.build_absolute_uri:'/' }}performance/api/?type=page_load&days=7"

# Track page view
curl -X POST "{{ request.build_absolute_uri:'/' }}performance/track/page-view/" \
  -H "Content-Type: application/json" \
  -d '{"url": "/", "load_time": 150, "title": "Home Page"}'

Search API

# Search all content
curl "{{ request.build_absolute_uri:'/' }}search/?q=cooperative"

# Search specific content type
curl "{{ request.build_absolute_uri:'/' }}search/?q=loan&type=services"

Contact API

# Submit contact form
curl -X POST "{{ request.build_absolute_uri:'/' }}contact/" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "John Doe",
    "email": "john@example.com",
    "subject": "Inquiry",
    "message": "Hello, I have a question about your services."
  }'
